\documentclass{resume}
\usepackage[utf8]{inputenc}
\usepackage[margin=1.5cm]{geometry}

% Slightly more compact type and spacing than a full CV.
\makeatletter
\renewcommand\normalsize{\@setfontsize\normalsize{10.5}{12.6}}
\normalsize
\makeatother
\setlength{\smallskipamount}{2pt plus 1pt minus 1pt}
\setlength{\medskipamount}{4pt plus 1pt minus 1pt}

\setfooter{Updated in Jan 2026}

\begin{document}

\name{Your Name}
\begin{center}
{\large Your Job Title \textperiodcentered\ Your Specialty}
\end{center}
\personalinfo{Your City}{Your Country}{}
             {your.email@example.com}{+1 (555) 123-4567}

\begin{center}\small
\href{https://github.com/yourusername}{GitHub}\ \textbar\
\href{https://yourwebsite.com}{Website}\ \textbar\
\href{https://www.linkedin.com/in/yourusername/}{LinkedIn}
\end{center}

%----------------------------------------------------------------------
\cvsection{Professional Summary}

\cvitemfull{One or two sentences summarizing your background, area of
expertise, and what you're looking for next. Keep it specific and
fact-based rather than full of generic buzzwords.}

%----------------------------------------------------------------------
\cvsection{Skills}

\cvitemfull{\textbf{Category One:} Skill, Skill, Skill, Skill, Skill.}
\cvitemfull{\textbf{Category Two:} Skill, Skill, Skill, Skill.}
\cvitemfull{\textbf{Category Three:} Skill, Skill, Skill.}

%----------------------------------------------------------------------
\cvsection{Experience}

\cveventfull{Jun 2023 -- Present}{Your Job Title}
            {Company Name}{City, Country}
\cvitemfull{Describe a key responsibility or achievement, ideally with a
measurable outcome.}
\cvitemfull{Describe another key responsibility or achievement.}

\cveventfull{Jan 2022 -- May 2023}{Previous Job Title}
            {Previous Company}{City, Country}
\cvitemfull{Describe a key responsibility or achievement.}

%----------------------------------------------------------------------
\cvsection{Education}

\cveventfull{Sep 2023 -- present}{Master of Science in Your Field}
            {Your University}{City, Country}
\cvitemfull{Cumulative GPA: 3.9/4.0}

\cveventfull{Sep 2018 -- Jun 2022}{Bachelor of Science in Your Field}
            {Your University}{City, Country}
\cvitemfull{Cumulative GPA: 3.6/4.0}

\end{document}
